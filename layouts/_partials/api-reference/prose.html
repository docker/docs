{{- $minimum := 6 -}}
{{- range findRE `]` $html -}}
  {{- $minimum = math.Min $minimum (int (substr . 2 1)) -}}
{{- end -}}
{{- $offset := sub (.level | default 3) $minimum -}}
{{- $levels := seq 6 -1 1 -}}
{{- if lt $offset 0 -}}{{- $levels = seq 1 6 -}}{{- end -}}
{{- range $levels -}}
  {{- $level := math.Min 6 (add . $offset) -}}
  {{- $html = replaceRE (printf `])` .) (printf `` .) (printf `` $level) -}}
{{- end -}}
