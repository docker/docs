{{- $name := . -}} {{- if strings.HasSuffix $name ".svg" -}} {{- $path := strings.TrimLeft "/" $name -}} {{- if not (strings.Contains $path "/") -}} {{- $path = printf "icons/%s" $path -}} {{- end -}} {{- with resources.Get $path -}} {{- .Content | safeHTML -}} {{- else -}} {{- errorf "Icon not found: %q (resolved to %q)" $name $path -}} {{- end -}} {{- else -}} {{- partialCached "ui-icon.html" (dict "name" $name) $name -}} {{- end -}}
